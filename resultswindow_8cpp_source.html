34void ResultsWindow::AppendResult(const ResultItem &resultitem, const std::string &key)
35{
36
37 result_item_counter++;
38 // The buttons find their result by its key in the Results map. This used
39 // to be rebuilt from the name, which went wrong for names containing a
40 // colon ("Convergence: split R-hat"): the lookup missed, created an empty
41 // result, and the table button crashed on it.
42 const QString result_key = QString::fromStdString(key);
43 QTextBrowser *textBrowser = new QTextBrowser(ui->scrollAreaWidgetContents);
44 textBrowser->setObjectName(result_key);
45 //textBrowser->setSizePolicy(QSizePolicy::Maximum,QSizePolicy::Maximum);
46 textBrowser->setTextColor(Qt::red);
47 textBrowser->append(QString::fromStdString(resultitem.Name())+":");
48 textBrowser->setTextColor(Qt::black);
49 textBrowser->setVerticalScrollBarPolicy(Qt::ScrollBarAlwaysOff);
50 QString str;
51 if (resultitem.ShowAsString())
52 str = QString::fromStdString(resultitem.Result()->ToString());
53 else
54 {
55 if (resultitem.ShowGraph() && resultitem.ShowTable())
56 str = QString::fromStdString("Push the graph or table button");
57 else if (resultitem.ShowGraph())
58 str = QString::fromStdString("Push the graph button");
59 else if (resultitem.ShowTable())
60 str = QString::fromStdString("Push the table button");
61
62 }
63
64 textBrowser->append(str);
65 int count = 0;
66 for(int i = 0;i < str.length();i++)
67 if(str.at(i).cell() == '\n')
68 count++;
69 textBrowser->setMaximumHeight(textBrowser->fontMetrics().height() * (count+3));
70 textBrowser->setMinimumHeight(textBrowser->fontMetrics().height() * (count+2));
71 ui->gridLayout->setAlignment(Qt::AlignTop);
72 ui->gridLayout->addWidget(textBrowser,ui->gridLayout->rowCount(),0);
73 if (resultitem.ShowGraph())
74 {
75 QPushButton *pushButtonGraph = new QPushButton(this);
76 QIcon iconGraph = QIcon(qApp->applicationDirPath()+"/../../resources/Icons/Graph.png");
77 pushButtonGraph->setIcon(iconGraph);
78 //pushButton->setSizePolicy(QSizePolicy::Minimum,QSizePolicy::Maximum);
79 pushButtonGraph->setMaximumWidth(20);
80 pushButtonGraph->setObjectName(result_key);
81
82 ui->gridLayout->addWidget(pushButtonGraph,ui->gridLayout->rowCount()-1,1,1,1,Qt::AlignTop);
83 connect(pushButtonGraph,SIGNAL(clicked()),this,SLOT(on_result_graph_clicked()));
84 }
85
86 QPushButton *pushButtonExport = new QPushButton(this);
87 QIcon iconExport = QIcon(qApp->applicationDirPath()+"/../../resources/Icons/export.png");
88 pushButtonExport->setIcon(iconExport);

92 ui->gridLayout->addWidget(pushButtonExport,ui->gridLayout->rowCount()-1,2,1,1,Qt::AlignTop);
93 connect(pushButtonExport,SIGNAL(clicked()),this,SLOT(on_result_export_clicked()));
94
95
96 if (resultitem.ShowTable())

110ResultItem* ResultsWindow::ResultForButton(QObject* button)
111{
112 if (results == nullptr || button == nullptr)
113 return nullptr;
114 // find(), not operator[]: a miss must not insert an empty result
115 auto it = results->find(button->objectName().toStdString());
116 if (it == results->end() || it->second.Result() == nullptr)
117 return nullptr;
118 return &it->second;
119}

134void ResultsWindow::on_result_table_clicked()
135{
136 ResultItem* item = ResultForButton(sender());

138 return;
139 ResultTableViewer *tableviewer = new ResultTableViewer(this);
140 tableviewer->setWindowFlag(Qt::WindowMinMaxButtonsHint);
141 QTableWidget *tablewidget = item->Result()->ToTable();
142 if (item->TableTitle() == "")
143 tableviewer->setWindowTitle(QString::fromStdString(item->Name()));
144 else
145 tableviewer->setWindowTitle(QString::fromStdString(item->TableTitle()));
146 tableviewer->SetTable(tablewidget);
147 tableviewer->show();
148}

150void ResultsWindow::on_result_export_clicked()
151{
152 ResultItem* item = ResultForButton(sender());
153 if (item == nullptr)
154 return;
155 QString fileName = QFileDialog::getSaveFileName(this,
156 tr("Save"), "",
157 tr("text file (*.txt)"));
158
159 if (!fileName.contains("."))
160 fileName+=".txt";
161 QFile file(fileName);
162
163 if (!file.open(QIODevice::WriteOnly | QIODevice::Text))
164 return;
165
166 item->Result()->writetofile(&file);
167
168 file.close();
169
170}
