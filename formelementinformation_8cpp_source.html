42void FormElementInformation::SyncIncludeAllCheckBox()
43{
44 ElementTableModel* model = static_cast<ElementTableModel*>(table()->model());
45 if (model == nullptr || model->Data == nullptr)
46 return;
47
48 bool all_included = true;
49 for (const auto& element : *model->Data->GetElementInformation())
50 {
51 if (!element.second.include_in_analysis)
52 {
53 all_included = false;
54 break;
55 }
56 }
57
58 // Reflect the data without acting on it
59 QSignalBlocker blocker(ui->Include_Exclude_All_checkBox);
60 ui->Include_Exclude_All_checkBox->setChecked(all_included);
61}

63void FormElementInformation::OnIncludeAllToggled()
64{
65 ElementTableModel* model = static_cast<ElementTableModel*>(table()->model());
66 if (model == nullptr || model->Data == nullptr)
67 return;
68
69 const bool include_all = (ui->Include_Exclude_All_checkBox->checkState() == Qt::CheckState::Checked);
70 model->Data->IncludeExcludeAllElements(include_all);
71 // Tell the view the "Include in analysis" column changed, so it shows the
72 // new values at once rather than when the table is next clicked
73 if (model->rowCount() > 0)
74 emit model->dataChanged(model->index(0, 4), model->index(model->rowCount() - 1, 4));
75 table()->viewport()->update();
76}
