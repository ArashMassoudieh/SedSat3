72/*/
73*@brief Copy assignment operator
74*
75* Assigns the contents of another profile to this one, replacing all
76* existing data.Handles self - assignment safely.
77*
78*@param other Profile to assign from
79* @return Reference to * this for chaining(e.g., a = b = c)
80*
81* @post* this contains copy of other's data
82* @post Previous contents of * this are replaced
83* @post Self - assignment(a = a) is safe
84*
85*@note Assignment order is important : Interface first, then members,
86* then base class to ensure proper cleanup if exceptions occur
87*/
88Elemental_Profile & Elemental_Profile::operator=(const Elemental_Profile & other)
89{
90 // Check for self-assignment (important for performance and correctness)
91 if (this == &other) {
92 return *this;
93 }
94
95 // Assign in proper order:
96 // 1. Interface base class
97 Interface::operator=(other);
98
99 // 2. Member variables
100 included_in_analysis_ = other.included_in_analysis_;
101 marked_elements_ = other.marked_elements_;
102
103 // 3. map base class (element concentrations)
104 // Done last because it might throw, and we want other data copied first
105 map<string, double>::operator=(other);
106
107 return *this;
108}

150Elemental_Profile Elemental_Profile::CreateAnalysisProfile(
151 const map<string, element_information>* element_info,
152 bool keep_isotope_bases) const
153{
154 // If no metadata provided, return complete copy
155 if (element_info == nullptr) {
156 return *this;
157 }
158
159 // Create filtered profile
160 Elemental_Profile filtered_profile;
161
162 // Copy analysis inclusion flag
163 filtered_profile.included_in_analysis_ = this->included_in_analysis_;
164
165 // With keep_isotope_bases (the datasets fingerprinting uses), the base
166 // elements of included isotopes are kept even when they are not tracers:
167 // an isotope's delta value is turned into a concentration with its base
168 // element's (e.g. delta 13C with TOC, the organic matter constituent).
169 // The tools that test or select tracers leave them out.
170 std::set<string> isotope_bases;
171 if (keep_isotope_bases)
172 for (const auto& [name, info] : *element_info)
173 if (info.Role == element_information::role::isotope && info.include_in_analysis && !info.base_element.empty())
174 isotope_bases.insert(info.base_element);
175
176 // Iterate through all elements in current profile
177 for (const auto& element_entry : *this) {
178 const string& element_name = element_entry.first;
179 const double concentration = element_entry.second;
180
181 // Check if element exists in metadata
182 auto info_iter = element_info->find(element_name);
183 if (info_iter == element_info->end()) {
184 // Element not in metadata - skip it
185 continue;
186 }
187
188 const element_information& info = info_iter->second;
189
190 // Apply inclusion criteria
191 bool should_include = (info.include_in_analysis &&
192 info.Role != element_information::role::do_not_include &&
193 info.Role != element_information::role::particle_size &&
194 info.Role != element_information::role::organic_carbon) ||
195 isotope_bases.count(element_name) > 0;
196
197 if (should_include) {
198 // Add element to filtered profile
199 filtered_profile[element_name] = concentration;
200
201 // Preserve marked status if element was marked
202 auto marked_iter = marked_elements_.find(element_name);
203 if (marked_iter != marked_elements_.end() && marked_iter->second) {
204 filtered_profile.marked_elements_[element_name] = true;
205 }
206 }
207 }
208
209 return filtered_profile;
210}

348Elemental_Profile Elemental_Profile::ExtractChemicalElements(
349 const map<string, element_information>* element_info,
350 bool include_isotopes) const
351{
352 // Validate input - element_info is required
353 if (element_info == nullptr) {
354 // For now, log error and return empty profile
355 // Future: throw std::invalid_argument("element_info cannot be null")
356 std::cerr << "Error: ExtractChemicalElements called with null element_info. "
357 << "Returning empty profile." << std::endl;
358 return Elemental_Profile();
359 }
360
361 Elemental_Profile extracted_profile;
362
363 // Iterate through all elements in current profile
364 for (const auto& [element_name, concentration] : *this) {
365
366 // Check if element exists in metadata
367 auto info_iter = element_info->find(element_name);
368 if (info_iter == element_info->end()) {
369 // Element not in metadata - skip it
370 // Could log warning: std::cerr << "Warning: Element '" << element_name
371 // << "' not found in metadata. Skipping.\n";
372 continue;
373 }
374
375 const element_information& info = info_iter->second;
376
377 // Check if element should be included based on role
378 bool should_include = false;
379
380 if (info.Role == element_information::role::element) {
381 // Always include chemical elements
382 should_include = true;
383 }
384 else if (info.Role == element_information::role::isotope) {
385 // Include isotopes only if requested
386 should_include = include_isotopes;
387 }
388 // All other roles (organic_carbon, particle_size, do_not_include) are excluded
389
390 if (should_include) {
391 extracted_profile[element_name] = concentration;
392 }
393 }
394
395 return extracted_profile;
396}

731 bool Elemental_Profile::ReadFromJsonObject(const QJsonObject& json_object, QString& error)
732 {
733 clear();
734
735 // Each value must be a finite number. (toDouble() turned a missing or
736 // non-numeric value, including a NaN saved as null, into 0.)
737 auto read_values = [&](const QJsonObject& values, bool skip_labels) {
738 for (const QString& key : values.keys()) {
739 if (skip_labels && (key == "XAxisLabel" || key == "YAxisLabel"))
740 continue;
741 double value = 0;
742 QString reason;
743 if (!parseNumericJson(values[key], value, reason)) {
744 error = QStringLiteral("constituent '%1': %2").arg(key, reason);
745 return false;
746 }
747 (*this)[key.toStdString()] = value;
748 }
749 return true;
750 };
751
752 if (json_object.contains("Include")) {
753 // Full format with metadata
754 SetIncludedInAnalysis(json_object["Include"].toBool());
755
756 if (!read_values(json_object["Element Contents"].toObject(), false))
757 return false;
758 }
759 else {
760 // Legacy format - elements directly in root
761 if (!read_values(json_object, true))
762 return false;
763 }
764
765 if (json_object.contains("XAxisLabel")) {
766 SetXAxisLabel(json_object["XAxisLabel"].toString());
767 }
768
769 if (json_object.contains("YAxisLabel")) {
770 SetYAxisLabel(json_object["YAxisLabel"].toString()); // Fixed: was SetXAxisLabel
771 }
772
773 return true;
774 }

920 Elemental_Profile Elemental_Profile::ApplyOrganicMatterAndSizeCorrections(
921 const vector<double>& reference_om_size,
922 const MultipleLinearRegressionSet* regression_models,
923 const map<string, element_information>* element_info) const
924 {
925 Elemental_Profile corrected_profile;
926
927 for (const auto& [element_name, concentration] : *this) {
928 const element_information& info = element_info->at(element_name);
929
930 // Organic carbon and particle size are not corrected. They are kept,
931 // unchanged, so that a later filter can keep the organic matter
932 // constituent when it is an isotope's base element (TOC for delta
933 // 13C); they used to be dropped here.
934 if (info.Role == element_information::role::organic_carbon ||
935 info.Role == element_information::role::particle_size) {
936 corrected_profile[element_name] = concentration;
937 continue;
938 }
939
940 // Start with original concentration
941 corrected_profile[element_name] = concentration;
942
943 // Apply corrections if regression model exists for this element
944 if (regression_models->count(element_name) == 0) {
945 continue;
946 }
947
948 const MultipleLinearRegression* mlr = &regression_models->at(element_name);
949 const auto& var_names = mlr->GetIndependentVariableNames();
950 const auto& coefficients = mlr->CoefficientsIntercept();
951 bool is_linear = (mlr->Equation() == regression_form::linear);
952
953 // Apply first correction (typically organic matter)
954 if (mlr->Effective(0) && var_names[0] != element_name) {
955 double reference = reference_om_size[0];
956 double measured = this->at(var_names[0]);
957
958 if (is_linear) {
959 corrected_profile[element_name] += (reference - measured) * coefficients[1];
960 }
961 else {
962 corrected_profile[element_name] *= pow(reference / measured, coefficients[1]);
963 }
964 }
965
966 // Apply second correction (typically particle size) if available
967 if (var_names.size() > 1 && mlr->Effective(1) && var_names[1] != element_name) {
968 double reference = reference_om_size[1];
969 double measured = this->at(var_names[1]);
970
971 if (is_linear) {
972 corrected_profile[element_name] += (reference - measured) * coefficients[2];
973 }
974 else {
975 corrected_profile[element_name] *= pow(reference / measured, coefficients[2]);
976 }
977 }
978
979 // Warn about negative corrected values (except for isotopes)
980 if (corrected_profile[element_name] < 0 &&
981 info.Role != element_information::role::isotope) {
982 std::cerr << "Warning: Corrected value for " << element_name
983 << " is negative: " << corrected_profile[element_name] << std::endl;
984 }
985 }
986
987 return corrected_profile;
988 }

1034 CMBVector Elemental_Profile::SortByConcentration(bool ascending) const
1035 {
1036 CMBVector sorted_result;
1037 vector<string> processed_elements;
1038
1039 for (size_t i = 0; i < size(); i++) {
1040 double target_value = ascending ? 1e6 : -1e6;
1041 string target_element;
1042
1043 for (const auto& [element_name, concentration] : *this) {
1044 // Skip already processed elements
1045 if (lookup(processed_elements, element_name) != -1) {
1046 continue;
1047 }
1048
1049 // Find min (ascending) or max (descending)
1050 bool is_better = ascending ? (concentration < target_value)
1051 : (concentration > target_value);
1052
1053 if (is_better) {
1054 target_element = element_name;
1055 target_value = concentration;
1056 }
1057 }
1058
1059 processed_elements.push_back(target_element);
1060 sorted_result.append(target_element, target_value);
1061 }
1062
1063 return sorted_result;
1064 }
