112inline bool ParseRandomSeed(const std::string& text, std::uint64_t& seed)
113{
114 std::size_t begin = text.find_first_not_of(" \t");
115 if (begin == std::string::npos)
116 {
117 seed = DrawRandomSeed();
118 return true;
119 }
120 std::size_t end = text.find_last_not_of(" \t");
121 const std::string digits = text.substr(begin, end - begin + 1);
122 if (digits.find_first_not_of("0123456789") != std::string::npos || digits.size() > 20)
123 return false;
124 try
125 {
126 std::size_t used = 0;
127 const unsigned long long value = std::stoull(digits, &used, 10);
128 if (used != digits.size())
129 return false;
130 seed = static_cast<std::uint64_t>(value);
131 return true;
132 }
133 catch (const std::exception&)
134 {
135 return false;
136 }
137}
