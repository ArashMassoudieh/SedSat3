34void IndicateSheetsDialog::Populate_Table(const QStringList &sheets)

38 QLabel *label = new QLabel(ui->scrollAreaWidgetContents);

43
44 QGroupBox *groupBox = new QGroupBox(ui->scrollAreaWidgetContents);
45 groupBox->setObjectName(QString::fromUtf8("groupBox"));

48 horizontalLayout->setObjectName(QString::fromUtf8("horizontalLayout"));
49 QRadioButton* radioButton_mixture = new QRadioButton(groupBox);
50 radioButton_mixture->setObjectName(QString::fromUtf8("Target"));
51 radioButton_mixture->setText("Target");
52 horizontalLayout->addWidget(radioButton_mixture);
53 radio_buttons_sinks.append(radioButton_mixture);
54 QRadioButton* radioButton_source = new QRadioButton(groupBox);
55 radioButton_source->setObjectName(QString::fromUtf8("Source"));
56 radioButton_source->setText("Source");
57 horizontalLayout->addWidget(radioButton_source);
58 radio_buttons_sources.append(radioButton_source);
59
60 radioButton_source->setChecked(i!=0);
61 radioButton_mixture->setChecked(i==0);
62 connect(radioButton_mixture,SIGNAL(clicked()),this, SLOT(on_radio_button_changed()));
63 connect(radioButton_source,SIGNAL(clicked()),this, SLOT(on_radio_button_changed()));
64 ui->formLayout->setWidget(i, QFormLayout::FieldRole, groupBox);
65
66 }
67 UpdateTargetHighlight();

70void IndicateSheetsDialog::UpdateTargetHighlight()
71{
72 QString target;
73 for (int i = 0; i < sheet_labels.count(); i++)
74 {
75 const bool is_target = radio_buttons_sinks[i]->isChecked();
76 sheet_labels[i]->setText(is_target ? group_names[i] + " (target)" : group_names[i]);
77 sheet_labels[i]->setStyleSheet(is_target ? "font-weight: bold; color: #0072B2;" : "");
78 if (is_target)
79 target = group_names[i];
80 }
81 if (summary_label != nullptr)
82 summary_label->setText(target.isEmpty()
83 ? tr("Choose the sheet that holds the target samples: select <b>Target</b> on its row. Every other sheet is a source group.")
84 : tr("Target samples: <b>%1</b>. Every other sheet is a source group.").arg(target.toHtmlEscaped()));
85}

91void IndicateSheetsDialog::accept()
92{
93 bool target_chosen = false;
94 for (QRadioButton* button : radio_buttons_sinks)
95 target_chosen = target_chosen || button->isChecked();
96 if (!target_chosen)
97 {
98 // Without a target sheet the import used to go ahead with none
99 QMessageBox::warning(this, tr("Target sheet"), tr("Select the sheet that holds the target samples."));
100 return;
101 }
102 for (int i=0; i<radio_buttons_sinks.count(); i++)
103 {
104 if (radio_buttons_sinks[i]->isChecked())
105 { dynamic_cast<MainWindow*>(parent())->SetSinkSheet(i);
106 dynamic_cast<MainWindow*>(parent())->Data()->SetTargetGroup(group_names[i].toStdString());
107 }
108 }
109 QDialog::accept();
110}

112void IndicateSheetsDialog::on_radio_button_changed()
113{

115 {
116 for (int i=0; i<radio_buttons_sinks.count(); i++)
117 {
118 if (radio_buttons_sinks[i]!=sender())
119 {
120 radio_buttons_sinks[i]->setChecked(false);
121 radio_buttons_sources[i]->setChecked(true);
122 }
123
124 }
125 }
126 UpdateTargetHighlight();
127}
