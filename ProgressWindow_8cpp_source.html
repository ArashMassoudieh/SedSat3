63bool ProgressWindow::HasChart(int chart) const
64{
65 // A window made without chart panels (number_of_panels == 0) ignores
66 // the chart calls instead of indexing past the end of ChartItems

81void ProgressWindow::AppendPoint(const double& x, const double& y, int chart)
82{
83 if (!HasChart(chart))
84 return;
85 if (ChartItems[chart].series->count()==0)
86 {
87 ChartItems[chart].yaxis->setMin(y-0.01);
88 ChartItems[chart].yaxis->setMax(y+0.01);
89 }
90 ChartItems[chart].series->append(x, y);
91 if (y<ChartItems[chart].yaxis->min())
92 ChartItems[chart].yaxis->setMin(y);
93 if (y>ChartItems[chart].yaxis->max())
94 ChartItems[chart].yaxis->setMax(y);
95 if (x<ChartItems[chart].xaxis->min())
96 ChartItems[chart].xaxis->setMin(x);
97 if (x>ChartItems[chart].xaxis->max())
98 ChartItems[chart].xaxis->setMax(x);
99 ChartItems[chart].chartView->update();
100
101}
