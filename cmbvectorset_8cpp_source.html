136double CMBVectorSet::FTest_p_value() const
137{
138 double SSB = 0;
139 double Overall_Mean = OverallMean();
140 for (map<string,CMBVector>::const_iterator vect = cbegin(); vect!=cend(); vect++ )
141 {
142 SSB += pow(vect->second.mean()-Overall_Mean,2)*vect->second.num;
143 }
144 double MSB = SSB/(size()-1);
145 double SSW=0;
146 for (map<string,CMBVector>::const_iterator vect = cbegin(); vect!=cend(); vect++ )
147 {
148 SSW += pow(vect->second.stdev(),2)*vect->second.num;
149 }
150 double MSW = SSW/(TotalNumberofObservations()-size());
151 double F = MSB/MSW;
152 double p_value = gsl_cdf_fdist_Q (F, size()-1, TotalNumberofObservations());
153 return p_value;
154}

156double CMBVectorSet::OverallMean() const
157{
158 double sum=0;
159 double count = 0;
160 for (map<string,CMBVector>::const_iterator vect = cbegin(); vect!=cend(); vect++ )
161 {

163 count+=vect->second.num;
