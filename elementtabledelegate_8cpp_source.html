17bool ElementTableDelegate::editorEvent(QEvent *event, QAbstractItemModel *model,
18 const QStyleOptionViewItem &option, const QModelIndex &index)
19{
20 // One click toggles "Include in analysis". It used to open a check box
21 // editor that needed another click, and that started ticked even for
22 // excluded elements (it read "No" as true).
23 if (index.column() == 4 && event->type() == QEvent::MouseButtonRelease)
24 {
25 const bool included = (index.data(Qt::DisplayRole).toString() == "Yes");
26 model->setData(index, !included, Qt::EditRole);
27 return true;
28 }
29 return QStyledItemDelegate::editorEvent(event, model, option, index);
30}

32QWidget *ElementTableDelegate::createEditor(QWidget *parent, const QStyleOptionViewItem &option,
33 const QModelIndex &index) const
34{
35 if (index.column() == 0) return QStyledItemDelegate::createEditor(parent, option, index);
36
37
38 if (index.column()==1)
39 {
40 QComboBox *editor = new QComboBox(parent);
41 editor->addItem("Element"); editor->addItem("Isotope"); editor->addItem("Particle Size");editor->addItem("Organic Carbon");editor->addItem("Exclude");
42 string element_name = index.sibling(index.row(),0).data(Qt::DisplayRole).toString().toStdString();
43 if (Data->GetElementInformation(element_name)->Role == element_information::role::do_not_include)
44 editor->setCurrentText("Exclude");
45 if (Data->GetElementInformation(element_name)->Role == element_information::role::element)
46 editor->setCurrentText("Element");
47 if (Data->GetElementInformation(element_name)->Role == element_information::role::isotope)
48 editor->setCurrentText("Isotope");
49 if (Data->GetElementInformation(element_name)->Role == element_information::role::particle_size)
50 editor->setCurrentText("Particle Size");
51 if (Data->GetElementInformation(element_name)->Role == element_information::role::organic_carbon)
52 editor->setCurrentText("Organic Carbon");
53
54
55 return editor;
56 }
57 if (index.column()==2)
58 {
59 QComboBox *editor = new QComboBox(parent);
60 vector<string> element_names = Data->GetElementNames();
61 if (index.sibling(index.row(),1).data(Qt::DisplayRole)!="Isotope")
62 { editor->setEnabled(false);
63 editor->setCurrentText("");
64 }
65 else
66 editor->setEnabled(true);
67
68
69 for (unsigned int i=0; i<element_names.size(); i++)
70 if (element_names[i]!=index.sibling(index.row(),0).data(Qt::DisplayRole).toString().toStdString())
71 editor->addItem(QString::fromStdString(element_names[i]));
72 editor->setCurrentText(index.data().toString());
73 return editor;
74 }
75 if (index.column()==3)
76 {
77 QLineEdit *editor = new QLineEdit(parent);
78 if (index.sibling(index.row(),1).data(Qt::DisplayRole)!="Isotope")
79 { editor->setEnabled(false);
80 editor->setText("");
81 }
82 else
83 editor->setEnabled(true);
84
85 QString text = index.data(Qt::DisplayRole).toString();
86
87 editor->setText(text);
88 return editor;
89 }
90 if (index.column()==4)
91 return nullptr; // toggled with one click in editorEvent()
92 if (index.column()==5)
93 return QStyledItemDelegate::createEditor(parent, option, index); // free-text notes
94
95 return nullptr;
96
97}
